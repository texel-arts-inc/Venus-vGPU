cmake_minimum_required(VERSION 3.24)
project(VenusVGPU VERSION 1.0 LANGUAGES C CXX)
include(cmake/FindExecutable.cmake)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build, for example cmake -S . -B build.")
endif()

option(VENUS_BUILD_HOST "Enable the macOS host app, runtime and packaging targets" ${APPLE})
option(VENUS_BUILD_GUEST "Enable the Linux aarch64 guest build targets" OFF)
option(VENUS_BUILD_DEMO "Build the Vulkan Cornell box" OFF)
option(VENUS_BUILD_OLLAMA "Build a native macOS Ollama runner for Venus Vulkan" OFF)

if(VENUS_BUILD_HOST)
    if(NOT APPLE)
        message(FATAL_ERROR "VENUS_BUILD_HOST requires macOS.")
    endif()
    include(cmake/VenusFramework.cmake)
    add_subdirectory(host)
endif()
if(VENUS_BUILD_GUEST)
    include(cmake/VenusUpstream.cmake)
    add_subdirectory(guest)
endif()
if(VENUS_BUILD_DEMO)
    add_subdirectory(demo)
endif()
if(VENUS_BUILD_OLLAMA)
    add_subdirectory(inference)
endif()
